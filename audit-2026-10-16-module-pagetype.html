可实施的共用政策参考与发布顺序

下例是已用合同样例验证的独立JavaScript参考函数，不是已修改或编译过的GARVEE生产代码。实际技术栈/版本、字段适配和组件路径须与前端仓库核对。输入必须先完成真实路由、鉴权、参数合法性和同快照结果集判断；国际映射来自受信任配置。

// Reference policy for validated routes and trusted configuration, not a live GARVEE patch.
export function seoPolicy(input) {
  const url = new URL(input.url);
  const hasQuery = [...url.searchParams.keys()].length > 0;
  let status = input.status ?? 200;
  if (status === 200 && input.dataFailed) status = 503;
  // Count only explicit user filters. Default inventory availability is a
  // separate lifecycle state; an established empty collection is not a 404.
  // zeroResults must come from a successful, authoritative result snapshot.
  if (status === 200 && (input.invalid ||
      (input.kind === 'collection' && input.hasUserFilters && input.zeroResults))) status = 404;
  if (status !== 200) {
    return { status, canonical: null, hreflang: {}, normalEntitySchema: false,
      robots: status === 404 || status === 410 ? 'noindex,follow' : null };
  }
  const isSearch = input.kind === 'search';
  const isPrivate = Boolean(input.privatePage);
  const noindex = hasQuery || isSearch || isPrivate;
  const query = input.normalizedQuery ?? url.searchParams.get('q') ?? '';
  // canonicalPath is supplied by the validated entity route registry.
  let canonical = input.canonicalPath
    ? new URL(input.canonicalPath, input.origin ?? 'https://www.garvee.com').href : null;
  if (isSearch) {
    canonical = query ? new URL('/search', input.origin ?? 'https://www.garvee.com') : null;
    if (canonical) {
      canonical.searchParams.set('q', query);
      canonical = canonical.href;
    }
  }
  const hreflang = !isSearch && !isPrivate && input.hreflangEligible
    ? (input.verifiedCanonicalAlternates ?? {}) : {};
  return { status, canonical, hreflang,
    robots: `${noindex ? 'noindex' : 'index'},${isPrivate ? 'nofollow' : 'follow'}`,
    normalEntitySchema: !isPrivate && !isSearch,
    // Add this partition to the full market/content/session-aware HTML cache key.
    policyPartition: hasQuery ? 'query-present' : 'query-absent' };
}
  1. 先修路由与政策：无参类目直接200、默认库存放服务端查询；参数统一noindex；分清零结果/非法/故障；同一routeDecision供Head、Schema、Sitemap及缓存使用。
  2. 再修Head与当前页主数据：补canonical/title/description、修OG身份与类型图片；导航、当前列表/分页、商品名价图与首批评论、文章正文按字段SSR。参数去canonical不丢真实功能状态。
  3. 按真实数据补Schema与语义：先商品/首页/类目/文章及footer实体，再验证国际对应与条件字段；清理无意义重复、错误main/ID/表单，不机械按标签数量改页面。
  4. 量测后改性能：先确认资源用途与关键路径，按页面类型按需加载；保留账户安全/必要业务能力。用实测定预算，现场P75与实验室各自验收。
  5. 按状态灰度与回滚：正常/参数/分页/零结果/错误/登录态/市场/生命周期回归；失效Sitemap与Feed同步；监控抓取和URL Inspection，缺权限保持未验证。本轮不承诺排名或流量提升。
